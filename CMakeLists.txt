cmake_minimum_required(VERSION 3.24)
project(OwnVPN VERSION 0.3.9 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(cmake/ZXing.cmake)
add_library(ownvpn_core src/core.cpp src/settings.cpp src/credential_store.cpp src/ipc.cpp src/profile.cpp src/protocol.cpp src/management_startup.cpp src/management_commands.cpp src/reconnect_policy.cpp src/totp.cpp src/totp_store.cpp src/totp_preview.cpp src/totp_detection.cpp)
target_sources(ownvpn_core PRIVATE src/dns_fallback.cpp src/ui_text.cpp)
target_include_directories(ownvpn_core PUBLIC src)
target_compile_definitions(ownvpn_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(ownvpn_core PUBLIC advapi32 crypt32 bcrypt shell32 ole32 ws2_32)
add_library(ownvpn_qr src/qr_import.cpp)
target_link_libraries(ownvpn_qr PUBLIC ownvpn_core ZXing::ZXing windowscodecs uuid)
target_compile_options(ownvpn_qr PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(OwnVPN WIN32 src/main.cpp src/system_tray.cpp src/usage_panel.cpp src/clipboard.cpp src/engine.cpp src/connection_controller.cpp src/otp_setup_dialog.cpp resources/app.rc)
target_link_libraries(OwnVPN PRIVATE ownvpn_core ownvpn_qr ws2_32 iphlpapi comctl32 comdlg32 uxtheme dwmapi)
target_compile_options(ownvpn_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_options(OwnVPN PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MINGW)
  target_link_options(OwnVPN PRIVATE -municode -static -static-libgcc -static-libstdc++)
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} -dumpspecs OUTPUT_VARIABLE GCC_SPECS)
  if(GCC_SPECS MATCHES "default-manifest")
    target_link_options(OwnVPN PRIVATE "-specs=${CMAKE_CURRENT_SOURCE_DIR}/resources/manifest.specs")
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(totp_diagnose tools/totp_diagnose.cpp)
  target_link_libraries(totp_diagnose PRIVATE ownvpn_core ownvpn_qr)
  target_compile_options(totp_diagnose PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(MINGW)
    target_link_options(totp_diagnose PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_executable(ownvpn_tests tests/tests.cpp src/engine.cpp)
  target_link_libraries(ownvpn_tests PRIVATE ownvpn_core ownvpn_qr ws2_32 iphlpapi)
  target_compile_definitions(ownvpn_tests PRIVATE OWNVPN_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_compile_options(ownvpn_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(MINGW)
    target_link_options(ownvpn_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core_and_management COMMAND ownvpn_tests)
  add_executable(totp_tests tests/totp_tests.cpp)
  target_link_libraries(totp_tests PRIVATE ownvpn_core ownvpn_qr)
  target_compile_definitions(totp_tests PRIVATE OWNVPN_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_compile_options(totp_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(MINGW)
    target_link_options(totp_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME totp_and_qr COMMAND totp_tests)
  add_executable(connection_tests tests/connection_tests.cpp src/connection_controller.cpp)
  target_link_libraries(connection_tests PRIVATE ownvpn_core)
  target_compile_options(connection_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(MINGW)
    target_link_options(connection_tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME disconnect_lifecycle COMMAND connection_tests)
  set_tests_properties(disconnect_lifecycle PROPERTIES TIMEOUT 25)
endif()
