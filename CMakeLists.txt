cmake_minimum_required(VERSION 3.24)
project(OwnVPN VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(OWNVPN_BUILD_WINDOWS_APP "Build Windows app and adapters" ${WIN32})
option(OWNVPN_BUILD_MACOS_APP "Build native macOS app and adapters" ${APPLE})
include(CTest)
function(ownvpn_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
function(ownvpn_static target)
  if(MINGW)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endfunction()
add_library(ownvpn_core STATIC
  src/core/core.cpp src/core/encoding.cpp src/core/unicode.cpp src/core/platform.cpp
  src/core/settings.cpp src/core/credential_store.cpp src/core/profile.cpp
  src/core/protocol.cpp src/core/management_startup.cpp src/core/management_commands.cpp
  src/core/reconnect_policy.cpp src/core/totp.cpp src/core/totp_store.cpp
  src/core/totp_preview.cpp src/core/totp_detection.cpp src/core/dns_fallback.cpp src/core/ui_text.cpp)
target_include_directories(ownvpn_core PUBLIC src/core)
ownvpn_warnings(ownvpn_core)
if(BUILD_TESTING)
  add_executable(portable_core_tests tests/portable_core_tests.cpp tests/support/test_platform.cpp)
  target_link_libraries(portable_core_tests PRIVATE ownvpn_core)
  ownvpn_warnings(portable_core_tests)
  ownvpn_static(portable_core_tests)
  add_test(NAME portable_core COMMAND portable_core_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME core_platform_boundary COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_core_boundary.py)
  endif()
endif()
if(OWNVPN_BUILD_MACOS_APP)
  if(NOT APPLE)
    message(FATAL_ERROR "macOS app needs an Apple SDK; use -DOWNVPN_BUILD_MACOS_APP=OFF for core")
  endif()
  enable_language(OBJCXX)
  add_subdirectory(src/platform/macos)
endif()
if(OWNVPN_BUILD_WINDOWS_APP)
  if(NOT WIN32)
    message(FATAL_ERROR "Windows app needs Windows toolchain; use -DOWNVPN_BUILD_WINDOWS_APP=OFF for core")
  endif()
  enable_language(RC)
  add_library(ownvpn_windows STATIC
    src/platform/windows/windows_services.cpp src/platform/windows/windows_crypto.cpp
    src/platform/windows/windows_store.cpp src/platform/windows/windows_runtime.cpp
    src/platform/windows/windows_network.cpp)
  target_include_directories(ownvpn_windows PUBLIC src/platform/windows)
  target_compile_definitions(ownvpn_windows PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(ownvpn_windows PUBLIC ownvpn_core advapi32 crypt32 bcrypt shell32 ole32 ws2_32 uuid)
  ownvpn_warnings(ownvpn_windows)
  include(cmake/ZXing.cmake)
  add_library(ownvpn_qr src/platform/windows/qr_import.cpp)
  target_link_libraries(ownvpn_qr PUBLIC ownvpn_windows ZXing::ZXing windowscodecs uuid)
  ownvpn_warnings(ownvpn_qr)
  add_executable(OwnVPN WIN32
    src/platform/windows/main.cpp src/platform/windows/system_tray.cpp
    src/platform/windows/usage_panel.cpp src/platform/windows/clipboard.cpp
    src/platform/windows/engine.cpp src/platform/windows/connection_controller.cpp
    src/platform/windows/otp_setup_dialog.cpp src/platform/windows/ipc.cpp resources/app.rc)
  target_link_libraries(OwnVPN PRIVATE ownvpn_core ownvpn_windows ownvpn_qr ws2_32 iphlpapi comctl32 comdlg32 uxtheme dwmapi)
  ownvpn_warnings(OwnVPN)
  ownvpn_static(OwnVPN)
  if(MINGW)
    target_link_options(OwnVPN PRIVATE -municode)
    execute_process(COMMAND ${CMAKE_CXX_COMPILER} -dumpspecs OUTPUT_VARIABLE GCC_SPECS)
    if(GCC_SPECS MATCHES "default-manifest")
      target_link_options(OwnVPN PRIVATE "-specs=${CMAKE_CURRENT_SOURCE_DIR}/resources/manifest.specs")
    endif()
  endif()
  if(BUILD_TESTING)
    add_executable(totp_diagnose tools/totp_diagnose.cpp)
    target_link_libraries(totp_diagnose PRIVATE ownvpn_core ownvpn_windows ownvpn_qr)
    add_executable(ownvpn_tests tests/tests.cpp src/platform/windows/engine.cpp src/platform/windows/ipc.cpp)
    target_link_libraries(ownvpn_tests PRIVATE ownvpn_core ownvpn_windows ownvpn_qr ws2_32 iphlpapi)
    target_compile_definitions(ownvpn_tests PRIVATE OWNVPN_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME core_and_management COMMAND ownvpn_tests)
    add_executable(totp_tests tests/totp_tests.cpp)
    target_link_libraries(totp_tests PRIVATE ownvpn_core ownvpn_windows ownvpn_qr)
    target_compile_definitions(totp_tests PRIVATE OWNVPN_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME totp_and_qr COMMAND totp_tests)
    add_executable(connection_tests tests/connection_tests.cpp src/platform/windows/connection_controller.cpp src/platform/windows/ipc.cpp)
    target_link_libraries(connection_tests PRIVATE ownvpn_core ownvpn_windows)
    add_test(NAME disconnect_lifecycle COMMAND connection_tests)
    set_tests_properties(disconnect_lifecycle PROPERTIES TIMEOUT 25)
    foreach(target totp_diagnose ownvpn_tests totp_tests connection_tests)
      ownvpn_warnings(${target})
      ownvpn_static(${target})
    endforeach()
    if(MINGW)
      target_link_options(totp_diagnose PRIVATE -municode)
      target_link_options(connection_tests PRIVATE -municode)
    endif()
  endif()
endif()
