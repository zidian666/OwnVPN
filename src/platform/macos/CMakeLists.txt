set(OWNVPN_MACOS_VERSION "0.6.0-alpha.1")
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <stop_token>\nint main(){std::stop_source s; return s.stop_requested();}" OWNVPN_MACOS_STOP_TOKEN)
if(NOT OWNVPN_MACOS_STOP_TOKEN)
  message(FATAL_ERROR "This Apple C++ library lacks std::stop_token. Use a current Xcode/Command Line Tools or a current LLVM libc++ toolchain.")
endif()
foreach(framework Foundation AppKit Security Vision ImageIO)
  find_library(${framework}_LIBRARY ${framework} REQUIRED)
endforeach()
find_library(DNS_SD_LIBRARY dns_sd REQUIRED)
find_library(ARCHIVE_LIBRARY archive REQUIRED)
add_library(ownvpn_macos STATIC mac_crypto.cpp mac_platform.mm mac_keychain.mm mac_network.cpp mac_qr.mm mac_import.cpp)
target_include_directories(ownvpn_macos PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ownvpn_macos PUBLIC ownvpn_core
  ${Foundation_LIBRARY} ${Security_LIBRARY} ${Vision_LIBRARY} ${ImageIO_LIBRARY} ${DNS_SD_LIBRARY} ${ARCHIVE_LIBRARY})
target_compile_options(ownvpn_macos PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
ownvpn_warnings(ownvpn_macos)
add_executable(OwnVPNMac MACOSX_BUNDLE mac_app.mm)
target_link_libraries(OwnVPNMac PRIVATE ownvpn_macos ${AppKit_LIBRARY})
target_compile_options(OwnVPNMac PRIVATE -fobjc-arc)
ownvpn_warnings(OwnVPNMac)
set_target_properties(OwnVPNMac PROPERTIES OUTPUT_NAME OwnVPN
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
  MACOSX_BUNDLE_BUNDLE_NAME OwnVPN MACOSX_BUNDLE_GUI_IDENTIFIER com.ownvpn.macos
  MACOSX_BUNDLE_BUNDLE_VERSION 600 MACOSX_BUNDLE_SHORT_VERSION_STRING ${OWNVPN_MACOS_VERSION})
if(BUILD_TESTING)
  add_executable(macos_platform_tests "${PROJECT_SOURCE_DIR}/tests/macos_platform_tests.mm")
  target_link_libraries(macos_platform_tests PRIVATE ownvpn_macos)
  target_compile_options(macos_platform_tests PRIVATE -fobjc-arc)
  target_compile_definitions(macos_platform_tests PRIVATE OWNVPN_TEST_FIXTURES="${PROJECT_SOURCE_DIR}/tests/fixtures")
  ownvpn_warnings(macos_platform_tests)
  add_test(NAME macos_platform COMMAND macos_platform_tests)
  set_tests_properties(macos_platform PROPERTIES TIMEOUT 30)
endif()
