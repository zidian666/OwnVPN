cmake_minimum_required(VERSION 3.24)
project(OwnVPNAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(OWNVPN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OWNVPN_BUILD_WINDOWS_APP OFF CACHE BOOL "" FORCE)
add_subdirectory("${OWNVPN_ROOT}" "${CMAKE_BINARY_DIR}/core")
include("${OWNVPN_ROOT}/cmake/AndroidDependencies.cmake")
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${mbedtls_SOURCE}" "${CMAKE_BINARY_DIR}/mbedtls" EXCLUDE_FROM_ALL)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${fmt_SOURCE}" "${CMAKE_BINARY_DIR}/fmt" EXCLUDE_FROM_ALL)
add_library(android_compression STATIC "${lz4_SOURCE}/lib/lz4.c" "${xxhash_SOURCE}/xxhash.c")
target_include_directories(android_compression PUBLIC "${lz4_SOURCE}/lib" "${xxhash_SOURCE}")
add_library(openvpn_android STATIC "${openvpn3_SOURCE}/client/ovpncli.cpp" "${openvpn3_SOURCE}/openvpn/crypto/data_epoch.cpp")
target_include_directories(openvpn_android SYSTEM PUBLIC "${openvpn3_SOURCE}" "${asio_SOURCE}/include")
target_compile_definitions(openvpn_android PUBLIC ASIO_STANDALONE USE_ASIO USE_MBEDTLS HAVE_LZ4)
target_link_libraries(openvpn_android PUBLIC mbedtls mbedx509 mbedcrypto android_compression fmt::fmt log)
include("${OWNVPN_ROOT}/cmake/ZXing.cmake")
add_library(ownvpn_android SHARED jni_support.cpp android_platform.cpp android_profiles.cpp android_vpn.cpp)
target_link_libraries(ownvpn_android PRIVATE ownvpn_core openvpn_android ZXing::ZXing android log)
target_compile_options(ownvpn_android PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(ownvpn_android PRIVATE "-Wl,-z,max-page-size=16384")
set(notices "${OWNVPN_ROOT}/build/android-licenses/licenses")
file(MAKE_DIRECTORY "${notices}")
configure_file("${OWNVPN_ROOT}/docs/android-third-party-notices.txt" "${notices}/NOTICE.txt" COPYONLY)
configure_file("${openvpn3_SOURCE}/LICENSES/MPL-2.0.txt" "${notices}/OpenVPN-MPL-2.0.txt" COPYONLY)
configure_file("${mbedtls_SOURCE}/LICENSE" "${notices}/MbedTLS.txt" COPYONLY)
configure_file("${asio_SOURCE}/COPYING" "${notices}/Asio-COPYING.txt" COPYONLY)
configure_file("${asio_SOURCE}/LICENSE_1_0.txt" "${notices}/Asio-Boost-1.0.txt" COPYONLY)
configure_file("${fmt_SOURCE}/LICENSE" "${notices}/fmt.txt" COPYONLY)
configure_file("${lz4_SOURCE}/lib/LICENSE" "${notices}/LZ4.txt" COPYONLY)
configure_file("${xxhash_SOURCE}/LICENSE" "${notices}/xxHash.txt" COPYONLY)
configure_file("${ZXING_SOURCE}/LICENSE" "${notices}/ZXing.txt" COPYONLY)
configure_file("${OWNVPN_ROOT}/THIRD_PARTY_NOTICES.txt" "${notices}/ZXing-embedded.txt" COPYONLY)
